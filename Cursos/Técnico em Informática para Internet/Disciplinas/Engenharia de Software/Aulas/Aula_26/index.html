Cálculo Desacoplado: calculadora_desconto.py 

class CalculadoraDescontoService:
    @staticmethod
    def calcular(subtotal: float, tipo: TipoDesconto, valor: float) -> float:
        if subtotal <= 0:
            return 0.0
        if tipo == TipoDesconto.PERCENTUAL:
            desconto = subtotal * (valor / 100.0)
        else:
            desconto = min(valor, subtotal)
        return round(desconto, 2)

Orquestrador Transacional: checkout_service.py 

def processar_checkout(self, pedido: Pedido, codigo: Optional[str] = None):
    if codigo:
        cupom = self.cupom_repo.buscar_por_codigo(codigo)
        if not cupom:
            raise CupomInexistenteError(codigo)
        pedido.aplicar_cupom(cupom)
    for item in pedido.itens:
        self.livro_repo.atualizar_estoque(item.livro.id, -item.quantidade)
    if pedido.cupom:
        self.cupom_repo.registrar_uso(pedido.cupom.id)
    pedido.status = "PAGO"
    return self.pedido_repo.salvar(pedido)

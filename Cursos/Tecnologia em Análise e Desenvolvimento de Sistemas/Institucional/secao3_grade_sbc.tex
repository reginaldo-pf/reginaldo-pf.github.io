        \begin{itemize}\setlength{\itemsep}{0.05em}
          \item Engenharia de Software \& Gestão de Projetos Ágeis
          \item Análise e Projeto de Sistemas (UML / Arquitetura)
          \item Interação Humano-Computador (Design UI/UX)
          \item Padrões de Projeto de Software (Design Patterns GoF)
        \end{itemize}
      \end{bluecard}
      
      \vspace{0.1cm}
      
      \begin{greencard}{\faServer\ 3. Infraestrutura para Software (400h)}
        \tiny
        \textbf{Competência:} Ambientes tecnológicos, redes e computação em nuvem.
        \begin{itemize}\setlength{\itemsep}{0.05em}
          \item Redes de Computadores \& Segurança da Informação
          \item Sistemas Operacionais \& Administração Linux
          \item Desenvolvimento e Operações (\textbf{DevOps \& CI/CD})
        \end{itemize}
      \end{greencard}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{purplecard}{\faCode\ 2. Implementação de Software (1.360h)}
        \tiny
        \textbf{Competência:} Programação avançada, bancos de dados e IA.
        \begin{itemize}\setlength{\itemsep}{0.05em}
          \item Lógica, POO e Estruturas de Dados Avançadas
          \item Bancos de Dados Relacionais (SQL) e Não-Relacionais (\textbf{NoSQL})
          \item Full-Stack Web (Web I e II) \& Apps Mobile nativos/híbridos
          \item \textbf{Inteligência Computacional \& Ciência de Dados}
          \item Testes Automatizados e Qualidade de Software
        \end{itemize}
      \end{purplecard}
      
      \vspace{0.1cm}
      
      \begin{moderncard}{\faUserFriends\ 4. Empreendedorismo \& Pessoal (480h)}
        \tiny
        \textbf{Competência:} Liderança, visão de negócio, comunicação e ética.
        \begin{itemize}\setlength{\itemsep}{0.05em}
          \item Empreendedorismo, Inovação e Modelos de Startups
          \item Inglês Técnico, Comunicação e Expressão
          \item Ética, Responsabilidade Socioambiental e Projeto Social
        \end{itemize}
      \end{moderncard}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Evolução Semestral da Formação}{\faStream\ Um Percurso Passo a Passo do Básico ao Nível Avançado}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{greencard}{\faSeedling\ Ciclo de Fundamentação (1º ao 3º Sem.)}
        \scriptsize
        \textbf{Construindo as Bases do Desenvolvedor:}
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{1º Semestre (400h):} Introdução à Programação, Banco de Dados, Tecnologias Web, Pensamento Computacional, Fundamentos de Matemática e Inglês Técnico.
          \item \textbf{2º Semestre (400h):} POO, Estrutura de Dados, Engenharia de Software, Redes de Computadores e Empreendedorismo.
          \item \textbf{3º Semestre (400h):} Programação Web I, Dispositivos Móveis, Bancos NoSQL, Análise e Projeto de Sistemas, Sistemas Operacionais e IHC.
        \end{itemize}
      \end{greencard}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{bluecard}{\faRocket\ Ciclo Avançado e Produção (4º e 5º Sem.)}
        \scriptsize
        \textbf{Inteligência, Nuvem e Prática Real:}
        \begin{itemize}\setlength{\itemsep}{0.2em}
          \item \textbf{4º Semestre (400h):} \textbf{Inteligência Computacional}, Programação Web II, Padrões de Projeto, Segurança da Informação e \textbf{PIM I} (Projeto Integrador).
          \item \textbf{5º Semestre (400h):} \textbf{DevOps}, Testes e Qualidade de Software, \textbf{PIM II}, Projeto Social e Optativas Avançadas.
          \item \textbf{Optativas de Destaque:} Ciência de Dados, Jogos Digitais, Tópicos em IA e Métodos Ágeis.
        \end{itemize}
      \end{bluecard}
    \end{column}
  \end{columns}

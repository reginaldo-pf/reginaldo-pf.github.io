% Secao 5: Diferenciais Praticos e Curricularizacao da Extensao

\begin{frame}{Diferenciais Práticos: PIM I \& II e PPS}{\faTools\ Da Teoria à Prática: Desenvolvendo Softwares Reais na Graduação}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{bluecard}{\faLayerGroup\ Projetos Integradores (PIM I e II)}
        \begin{itemize}\setlength{\itemsep}{0.35em}\footnotesize
          \item \textbf{Integração Multidisciplinar:} Componentes obrigatórios no 4º e 5º semestres com 160h combinadas.
          \item \textbf{Criação de Produto Completo:} As equipes concebem, projetam, codificam, testam e implantam um software funcional.
          \item \textbf{Portfólio para Recrutadores:} O estudante se forma com sistemas reais publicados no GitHub e em produção para apresentar em entrevistas de emprego.
        \end{itemize}
      \end{bluecard}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{greencard}{\faUserCheck\ Prática Profissional Supervisionada (PPS)}
        \begin{itemize}\setlength{\itemsep}{0.35em}\footnotesize
          \item \textbf{40 Horas de Vivência de Mercado:} Distribuídas diretamente dentro de PIM I e PIM II (20h em cada).
          \item \textbf{Ambiente Simulado de Software House:} Metodologias ágeis, reuniões diárias (dailies), revisões de sprint e entrega de valor.
          \item \textbf{Supervisão Docente Direta:} Feedback contínuo de professores mestres e doutores com experiência na indústria.
        \end{itemize}
      \end{greencard}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Curricularização da Extensão: Impacto Social}{\faHandsHelping\ 10\% do Curso Dedicado à Transformação da Comunidade Local}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{greencard}{\faHandHoldingHeart\ O que é a Extensão Curricularizada?}
        \begin{itemize}\setlength{\itemsep}{0.35em}\footnotesize
          \item \textbf{210 Horas Obrigatórias (10\% da CH Total):} Integradas progressivamente do 2º ao 5º semestre.
          \item \textbf{Diálogo com a Sociedade:} Ações gratuitas e diretas que conectam a sala de aula às dores reais da população.
          \item \textbf{Disciplinas Envolvidas:} Empreendedorismo, Ética, Dispositivos Móveis, PIM I e II, e Projeto Social.
        \end{itemize}
      \end{greencard}
    \end{column}
    
    \begin{column}{0.48\textwidth}
      \begin{bluecard}{\faCity\ Impacto nos Sertões dos Inhamuns}
        \begin{itemize}\setlength{\itemsep}{0.35em}\footnotesize
          \item \textbf{Soluções para o Comércio \& Produtores:} Criação de sistemas de gestão, apps e canais digitais para arranjos produtivos locais.
          \item \textbf{Inclusão Digital \& Cidadania:} Oficinas gratuitas de capacitação digital para jovens, escolas públicas e agricultores.
          \item \textbf{Desenvolvimento de Soft Skills:} O discente aprimora empatia, oratória, escuta ativa, liderança e compromisso ético-social.
        \end{itemize}
      \end{bluecard}
    \end{column}
  \end{columns}
\end{frame}

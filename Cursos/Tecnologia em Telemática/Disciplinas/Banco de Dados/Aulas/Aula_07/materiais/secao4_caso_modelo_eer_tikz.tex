\twocolframe[title={Entidade Fraca e Especialização EER na Clínica ``VidaPlena''},leftcol=7.0cm,rightcol=7.0cm]{%
  \alert{Entidade Fraca: \texttt{DEPENDENTE}}\par
  \itemseps[topsepi=0.5mm,itemsepi=1.2mm,labelsep=2mm,leftmargini=3mm]
  \begin{itemize}
    \item \textbf{Entidade Proprietária:} \texttt{PACIENTE} (\underline{id\_prontuario}).
    \item \textbf{Relacionamento Identificador:} \texttt{POSSUI\_DEP} (Losango Duplo, $1:N$).
    \item \textbf{Discriminador (Chave Parcial):} \dashuline{nome\_dep} (Sublinhado Tracejado).
    \item \textbf{Chave Composta Formada:} $(\text{id\_prontuario}, \text{nome\_dep})$. Dois pacientes distintos podem ter dependentes chamados ``Lucas'' sem colisão!
  \end{itemize}
}{%
  \alert{Hierarquia EER: \texttt{COLABORADOR}}\par
  \itemseps[topsepi=0.5mm,itemsepi=1.2mm,labelsep=2mm,leftmargini=3mm]
  \begin{itemize}
    \item \textbf{Superclasse:} \texttt{COLABORADOR} (atributos universais: \underline{matricula}, \texttt{nome}, \texttt{endereco}, \texttt{telefones}).
    \item \textbf{Subclasses:} \texttt{MEDICO} (\texttt{crm}, \texttt{especialidades}) e \texttt{ATENDENTE} (\texttt{ramal}, \texttt{turno}).
    \item \textbf{Restrições EER:}
      \begin{itemize}\scriptsize
        \item \textbf{Disjunta ($d$):} Médico $\cap$ Atendente $= \emptyset$.
        \item \textbf{Total (Linha Dupla):} Todo colaborador pertence a uma das subclasses.
      \end{itemize}
  \end{itemize}
}

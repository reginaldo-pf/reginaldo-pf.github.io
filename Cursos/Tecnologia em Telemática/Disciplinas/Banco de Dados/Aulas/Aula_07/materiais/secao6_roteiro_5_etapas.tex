  \begin{itemize}
    \item \textbf{Etapa 1 (2,0 pts) -- Narrativa do Minimundo:}
      Título do projeto e texto corrido (15 a 20 linhas) detalhando todas as regras de negócio.
    \item \textbf{Etapa 2 (2,0 pts) -- Entidades e Atributos:}
      Tabela com $\ge 5$ entidades fortes (e PKs) + destaque do atributo composto, multivalorado e derivado.
    \item \textbf{Etapa 3 (2,0 pts) -- Matriz de Relacionamentos:}
      Especificação formal de $1:1$, $1:N$, $N:M$ (com atributo) e Unário/Recursivo (com papéis e $(min,max)$).
  \end{itemize}

\begin{lstlisting}[language=Python,basicstyle=\ttfamily\fontsize{5.2}{6.2}\selectfont]
class PortaInterface:
    def __init__(self, numero_porta, tipo_midia, velocidade_mbps):
        # Discriminador (Chave Parcial)
        self.numero_porta = numero_porta
        self.tipo_midia = tipo_midia
        self.velocidade_mbps = velocidade_mbps
        self.dispositivo_proprietario = None

    @property
    def chave_completa(self):
        if not self.dispositivo_proprietario:
            return None
        # PK Composta: (PK_Proprietario, Discriminador)
        return (self.dispositivo_proprietario.ip_gestao, 
                self.numero_porta)
\end{lstlisting}

class Roteador(DispositivoGenerico):
    def __init__(self, pat, fab, mod, rotas, bgp):
        super().__init__(pat, fab, mod)
        self.tabela_rotas_max = rotas
        self.suporta_bgp = bgp

def especializar(validador, pat, subclasse):
    subs = validador.classificacoes[pat]
    if validador.disjuncao == 'd' and len(subs) > 0:
        raise ValueError(f"Violacao (d): {pat} ja e {list(subs)[0]}!")
    subs.add(subclasse)

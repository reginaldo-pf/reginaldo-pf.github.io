1. O Ciclo de Vida do Projeto de Banco de Dados

O desenvolvimento profissional de bancos de dados é governado por um processo de engenharia dividido em quatro fases sequenciais e iterativas. Essa segregação garante que a modelagem semântica das regras de negócio permaneça protegida de detalhes e limitações de implementação em disco:

1. Requisitos

Entrevistas e análise do Minimundo. Gera especificação textual de necessidades de dados e operações.

2. Conceitual

Esquema semântico de alto nível via MER (Peter Chen, 1976). Totalmente independente de qualquer SGBD.

3. Lógico

Mapeamento para o paradigma relacional. Tabelas, chaves primárias/estrangeiras e Normalização (1FN a 3FN).

4. Físico

Implementação no SGBD alvo (PostgreSQL). Estruturas em disco, índices B-Tree e partições.

In [1]: Metamodelo Conceitual do Modelo Entidade-Relacionamento em Python 
from enum import Enum
from datetime import datetime, date
from typing import Any, Dict, List, Optional, Callable

class TipoAtributo(Enum):
    IDENTIFICADOR = "Chave Identificadora (Sublinhado no DER)"
    SIMPLES = "Simples / Atômico (Elipse simples no DER)"
    COMPOSTO = "Composto (Elipse ramificada no DER)"
    MULTIVALORADO = "Multivalorado (Elipse dupla no DER)"
    DERIVADO = "Derivado (Elipse tracejada no DER)"

class DefinicaoAtributo:
    def __init__(self, nome: str, tipo: TipoAtributo, descricao: str,
                 sub_atributos: Optional[List['DefinicaoAtributo']] = None,
                 funcao_derivacao: Optional[Callable[[Dict[str, Any]], Any]] = None,
                 obrigatorio: bool = True):
        self.nome = nome
        self.tipo = tipo
        self.descricao = descricao
        self.sub_atributos = sub_atributos or []
        self.funcao_derivacao = funcao_derivacao
        self.obrigatorio = obrigatorio

class EsquemaEntidade:
    def __init__(self, nome: str, descricao: str):
        self.nome = nome
        self.descricao = descricao
        self.atributos: Dict[str, DefinicaoAtributo] = {}
        self.instancias: List[Dict[str, Any]] = []

    def adicionar_atributo(self, atributo: DefinicaoAtributo):
        self.atributos[atributo.nome] = atributo

    def validar_e_inserir(self, dados_instancia: Dict[str, Any]):
        # 1. Validação de Unicidade do Identificador Conceitual
        id_attr = next((a for a in self.atributos.values() if a.tipo == TipoAtributo.IDENTIFICADOR), None)
        if id_attr:
            val_id = dados_instancia.get(id_attr.nome)
            if val_id is None:
                raise ValueError(f"Chave identificadora '{id_attr.nome}' não pode ser nula.")
            for inst in self.instancias:
                if inst.get(id_attr.nome) == val_id:
                    raise ValueError(f"Violação de unicidade: A chave '{val_id}' já está em uso.")

        # 2. Computação Dinâmica de Atributos Derivados
        for nome_attr, attr in self.atributos.items():
            if attr.tipo == TipoAtributo.DERIVADO and attr.funcao_derivacao:
                dados_instancia[nome_attr] = attr.funcao_derivacao(dados_instancia)

        self.instancias.append(dados_instancia)

print("✅ Metamodelo do MER carregado com sucesso.")

2. Taxonomia de Atributos de Peter Chen (1976)

No Modelo Entidade-Relacionamento, os atributos qualificam as entidades e são formalmente classificados de acordo com três dimensões estruturais independentes:

  • Simples (Atômico) vs. Composto: Atributos simples são indivisíveis (ex: hostname, mtu); compostos formam hierarquias menores com significado próprio (ex: localizacao_rack $\to$ bloco, sala, rack, unidade U).
  • Monovalorado vs. Multivalorado: Monovalorados contêm um único valor fixo (ex: mac_address); multivalorados formam coleções de valores para a mesma entidade (ex: conjunto de vlan_tags).
  • Armazenado vs. Derivado: Armazenados são fisicamente gravados no banco (ex: data_instalacao); derivados são calculados dinamicamente via expressão (ex: dias_operacao $= \text{hoje} - \text{data\_instalacao}$).
  • Identificador (Chave Primária): Atributo cujos valores identificam de forma única e inequívoca cada entidade individual no conjunto.

# Instanciação do Conjunto de Entidades (Entity Set)
esquema = EsquemaEntidade(
    nome="DISPOSITIVO_REDE",
    descricao="Ativos de rede da infraestrutura de telecomunicações do campus."
)

# 1. Atributo Identificador (Chave Primária no DER)
esquema.adicionar_atributo(DefinicaoAtributo("patrimonio_id", TipoAtributo.IDENTIFICADOR, "ID patrimonial único gravado no chassi"))

# 2. Atributos Simples / Atômicos
esquema.adicionar_atributo(DefinicaoAtributo("hostname", TipoAtributo.SIMPLES, "Nome FQDN na topologia"))
esquema.adicionar_atributo(DefinicaoAtributo("ip_loopback", TipoAtributo.SIMPLES, "IPv4 de gerência loopback0"))
esquema.adicionar_atributo(DefinicaoAtributo("fabricante", TipoAtributo.SIMPLES, "Fabricante do equipamento"))
esquema.adicionar_atributo(DefinicaoAtributo("data_instalacao", TipoAtributo.SIMPLES, "Data de ativação física (armazenado)"))

# 3. Atributo Composto (Hierárquico com subcomponentes atômicos)
sub_rack = [
    DefinicaoAtributo("bloco", TipoAtributo.SIMPLES, "Bloco predial"),
    DefinicaoAtributo("sala_noc", TipoAtributo.SIMPLES, "Sala técnica"),
    DefinicaoAtributo("rack_id", TipoAtributo.SIMPLES, "Identificador do rack"),
    DefinicaoAtributo("posicao_u", TipoAtributo.SIMPLES, "Unidade de altura no rack")
]
esquema.adicionar_atributo(DefinicaoAtributo("localizacao_rack", TipoAtributo.COMPOSTO, "Posicionamento no datacenter", sub_atributos=sub_rack))

# 4. Atributo Multivalorado (Conjunto de tags)
esquema.adicionar_atributo(DefinicaoAtributo("vlan_tags", TipoAtributo.MULTIVALORADO, "Conjunto de VLANs IEEE 802.1Q suportadas"))

# 5. Atributo Derivado (Calculado dinamicamente)
def calcular_dias_operacao(dados: Dict[str, Any]) -> int:
    dt_inst = datetime.strptime(dados["data_instalacao"], "%Y-%m-%d").date()
    return (date(2026, 10, 6) - dt_inst).days

esquema.adicionar_atributo(DefinicaoAtributo(
    nome="dias_operacao",
    tipo=TipoAtributo.DERIVADO,
    descricao="Tempo em dias decorrido desde a instalação",
    funcao_derivacao=calcular_dias_operacao,
    obrigatorio=False
))

print(f"✅ Esquema conceitual '{esquema.nome}' concebido com {len(esquema.atributos)} atributos.")

In [4]: Mapeamento Pré-Relacional no SQLite: O Papel da 1ª Forma Normal (1FN) 
import sqlite3

conn = sqlite3.connect(":memory:")
cursor = conn.cursor()

# 1. Tabela Principal (Atributo Composto achatado em colunas atômicas)
cursor.execute("""
CREATE TABLE dispositivos_rede (
    patrimonio_id TEXT PRIMARY KEY,
    hostname TEXT NOT NULL UNIQUE,
    ip_loopback TEXT NOT NULL UNIQUE,
    data_instalacao DATE NOT NULL,
    rack_bloco TEXT NOT NULL,
    rack_sala TEXT NOT NULL,
    rack_id TEXT NOT NULL,
    rack_posicao_u TEXT NOT NULL
);
""")

# 2. Tabela Auxiliar para o Atributo Multivalorado (Preservando a 1FN de Codd)
cursor.execute("""
CREATE TABLE dispositivo_vlans (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    dispositivo_patrimonio TEXT NOT NULL,
    vlan_id INTEGER NOT NULL CHECK (vlan_id BETWEEN 1 AND 4094),
    FOREIGN KEY (dispositivo_patrimonio) REFERENCES dispositivos_rede(patrimonio_id) ON DELETE CASCADE
);
""")

# 3. Visão (VIEW) para o Atributo Derivado
cursor.execute("""
CREATE VIEW vw_dispositivos_uptime AS
SELECT 
    patrimonio_id, hostname, ip_loopback, rack_id, data_instalacao,
    CAST((julianday('2026-10-06') - julianday(data_instalacao)) AS INTEGER) AS dias_operacao
FROM dispositivos_rede;
""")

# Inserção de dados
cursor.execute("INSERT INTO dispositivos_rede VALUES ('IFCE-TAU-00101', 'TAU-RTR-CORE-01', '10.200.0.1', '2024-02-15', 'Principal', 'NOC 102', 'RACK-01', 'U40-U42');")
cursor.executemany("INSERT INTO dispositivo_vlans (dispositivo_patrimonio, vlan_id) VALUES (?, ?);", [
    ('IFCE-TAU-00101', 10), ('IFCE-TAU-00101', 20), ('IFCE-TAU-00101', 100)
])
conn.commit()

# Consulta DQL com reconstituição do multivalorado via GROUP_CONCAT
cursor.execute("""
SELECT d.hostname, d.dias_operacao, GROUP_CONCAT(v.vlan_id, ', ') AS vlans
FROM vw_dispositivos_uptime d
JOIN dispositivo_vlans v ON d.patrimonio_id = v.dispositivo_patrimonio
GROUP BY d.patrimonio_id;
""")
for row in cursor.fetchall():
    print(f"-> Dispositivo: {row[0]:<16} | Uptime (Derivado): {row[1]} dias | VLANs (Multivalorado): [{row[2]}]")
